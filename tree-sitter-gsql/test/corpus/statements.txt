CREATE QUERY q(INT x) {
  INT ifCount = 0;
  IF x > 1 THEN
    ifCount = 1;
  ELSE // the comment does not split ELSE IF
    IF x > 0 THEN
      ifCount = 2;
    ELSE
      ifCount = 3;
  END;
  IF x > 1 THEN ifCount = 1; ELSE ifCount = 2; END;
}

  SumAccum<INT> @m;
  R = SELECT s FROM P:s
      POST-ACCUM
        FOREACH i IN RANGE[1, 2] DO
          DOUBLE nd = 0,
          IF a THEN nd = 1
          ELSE IF b THEN nd = 2
          ELSE nd = 3 END END,
          s.@m += nd
        END;
  IF a THEN PRINT 1; ELSE IF b THEN PRINT 2; ELSE PRINT 3; END; END;
}
