CREATE SCHEMA_CHANGE JOB j1 FOR GRAPH Social {
  ADD VERTEX Cat (PRIMARY_ID name STRING, age INT);
  ADD UNDIRECTED EDGE Owns (FROM Person, TO Cat);
  ALTER VERTEX Person ADD ATTRIBUTE (email STRING DEFAULT "none", score FLOAT);
  ALTER VERTEX Person DROP ATTRIBUTE (email);
  ALTER VERTEX Person ADD INDEX name_idx ON (name);
  ALTER VERTEX Person DROP INDEX name_idx;
  ALTER EDGE Owns ADD PAIR (FROM Person, TO Dog);
  ALTER VERTEX Person WITH TAGGABLE="true";
  DROP VERTEX Dog;
  DROP EDGE Has, Had;
}
RUN SCHEMA_CHANGE JOB j1
