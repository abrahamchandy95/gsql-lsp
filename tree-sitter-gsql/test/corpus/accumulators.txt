CREATE QUERY accums() {
  SumAccum<INT> @@total = 0, @@other;
  SumAccum<STRING COMPRESS> @names;
  MaxAccum<DOUBLE> @@max;
  AvgAccum @avg;
  OrAccum @visited = FALSE;
  BitwiseAndAccum @@bits;
  ListAccum<STRING> @@l = ["a", "b"];
  SetAccum<VERTEX<Person>> @@people;
  SetAccum<EDGE> @@edges;
  BagAccum<INT> @@bag = (1, 2, 2);
  MapAccum<STRING, SumAccum<INT>> @@counts;
  MapAccum<INT, MapAccum<STRING, SetAccum<VERTEX<Person>>>> @@nested;
  HeapAccum<Rec>(10, score DESC, name ASC) @@heap;
  GroupByAccum<INT a, STRING b, SumAccum<INT> c, MaxAccum<FLOAT> m> @@group;
  ArrayAccum<SumAccum<INT>> @@arr[5][10];
  STATIC ListAccum<INT> @@persisted;
}

CREATE QUERY updates() {
  @@total += 1;
  @@counts += ("a" -> 1);
  @@group += (1, "a" -> 2, 3.5);
  @@arr[1][2] += 5;
  @@heap += Rec(1, "x");
  @@l.clear();
  x = @@heap.top().score;
}

CREATE QUERY more_accums(INT len) {
  TYPEDEF HeapAccum <Rec> (2, name DESC, age ASC) My_Heap;
  My_Heap @@heap;
  HeapAccum<Rec>(score DESC) @@unbounded;
  HeapAccum<Rec>(k, score DESC) @@top_k;
  SumAccum<INT> EDGE @weight = 1;
  DeviationAccum @@dev;
  DeviationPAccum @@devp;
  BitwiseOrAccum<128> @@wide;
  BitwiseAndAccum<len> @@sized;
  ListAccum<TUPLE<INT, STRING>> @@pairs;
}
