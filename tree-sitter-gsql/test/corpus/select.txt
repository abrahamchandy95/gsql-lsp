CREATE QUERY q() {
  Result = SELECT t FROM Start:s -(Friendship:e)- Person:t
           SAMPLE 10 EDGE WHEN s.outdegree() > 10
           WHERE t.age > 18 AND s != t
           ACCUM t.@count += 1, @@total += 1
           POST-ACCUM t.@count = t.@count * 2
           POST_ACCUM @@x += t.@count
           HAVING t.@count > 1
           ORDER BY t.@count DESC, t.name ASC
           LIMIT 10;
}
